cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(FramegrabberPresentation LANGUAGES CXX)
endif()
if(TARGET FramegrabberPresentation)
    return()
endif()
if(TARGET Qt6::Widgets)
    set(presentation_qt Qt6)
elseif(TARGET Qt5::Widgets)
    set(presentation_qt Qt5)
else()
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    set(presentation_qt Qt6)
endif()
add_library(FramegrabberPresentation STATIC FramegrabberControlUi.cpp FramegrabberControlUi.h)
set_target_properties(FramegrabberPresentation PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
target_compile_features(FramegrabberPresentation PUBLIC cxx_std_17)
target_compile_options(FramegrabberPresentation PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_include_directories(FramegrabberPresentation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(FramegrabberPresentation PUBLIC ${presentation_qt}::Widgets)
